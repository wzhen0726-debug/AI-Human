基于调研报告中的测试数据与根因分析，当前问题的关键在于：QuadRemesher (QR) 均匀拓扑消解了几何曲率，导致基于角度/法线的常规自动缝合线算法全面失效，进而产生数千个单面碎片岛。

针对你的 **Blender 5.1 后台（`--background`）无 GUI 自动化控制** 需求，这里提供 3 个可直接落地的**脚本级解决方案与代码逻辑**，解决现有的“Edge ID 错位”、“碎片岛过多”和“人工标记接缝”问题。

## 方案一：修复 RizomUV LUA 管线（彻底解决 Edge ID 错位与拉伸）

### 1. 核心突破逻辑

在报告中，RizomUV 效果差（2.5/10）的主要原因在于 FBX 导出重排了 Edge ID。
**解决方案**：不要向 RizomUV 传递 Edge ID 数组！先在 Blender 内对缝合线做一次基础 Unwrap（哪怕拉伸极重），导出 FBX 后，利用 RizomUV LUA 内置的 `UVBorders` 选择器提取边界，再重新解算。

### 2. Blender 端导出脚本 (`scripts/export_for_rizom.py`)

```python
import bpy

def prepare_mesh_for_rizom(obj_name, output_fbx_path):
    obj = bpy.data.objects[obj_name]
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.mode_set(mode='EDIT')
    
    # 1. 如果已有缝合线 (Mark Seam)，强制进行一次基础展开，将 Seam 转化为 UV 岛边界
    bpy.ops.uv.unwrap(method='ANGLE_BASED', margin=0.001)
    bpy.ops.object.mode_set(mode='OBJECT')
    
    # 2. 导出 FBX (确保 UV 数据被导出)
    bpy.ops.export_scene.fbx(
        filepath=output_fbx_path,
        use_selection=True,
        mesh_smooth_type='FACE',
        add_leaf_bones=False
    )

prepare_mesh_for_rizom("03_remesh", "v5_run/in_for_rizom.fbx")

```

### 3. RizomUV 自动化 LUA 脚本 (`v5_run/auto_unwrap.lua`)

```lua
-- 1. 加载模型并重置基础 UV
ZomLoad({File={Path="v5_run/in_for_rizom.fbx", ImportGroups=true, XYZ=true}})

-- 2. 核心：直接选择 Blender 传过来的 UV 边界线 (即 Mark Seam 处)
ZomSelect({PrimType="Edge", WorkingSet="UVBorders", ResetBefore=true})

-- 3. 转换为 3D 切割缝合线
ZomCut({PrimType="Edge", WorkingSet="Selected"})

-- 4. 执行工业级 Unfold 与 Stretch 消除 (彻底解决 2.5分 的拉伸问题)
ZomUnfold({PrimType="Polygon", WorkingSet="All", Iterations=20, PinIslandCenters=false})
ZomOptimize({Iterations=15, Method="Mosaic"})

-- 5. 自动缩放像素密度与高利用率 Packing
ZomIslandGroup({Mode="Set", WorkingSet="All", PackProperties={AlignMode=1}})
ZomPack({ProcessTile=0, MarginVisible=true, MarginPS=0.002, Rotate={Mode=2, Step=90}})

-- 6. 保存并退出
ZomSave({File={Path="v5_run/out_from_rizom.fbx"}})
ZomQuit()

```

## 方案二：Blender 内部碎片岛后处理消除脚本 (消除 2892 个单面碎片)

如果你希望继续沿用 **ZEN UV (8.25/10 分方案)**，目前唯一的缺点是 **存在约 2000 个 1-Quad 单面碎片岛**。可以通过以下 Python 后处理脚本，在 Blender 后台**自动识别并合并碎片岛到最近的主岛**：

```python
import bpy
import bmesh

def cleanup_fragment_uv_islands(obj, min_faces_per_island=5):
    """自动清理 QuadRemesher 产生的单面碎片岛"""
    bpy.ops.object.mode_set(mode='EDIT')
    bm = bmesh.from_edit_mesh(obj.data)
    uv_layer = bm.loops.layers.uv.verify()
    
    # 1. 查找所有独立 UV 岛及其面数量 (后台算法)
    # 算法逻辑：递归遍历连通的 Loop，统计每个岛的面数
    # 对面数 < min_faces_per_island 的碎片岛，清除其边缘 seam 并合并到相邻主岛
    
    bpy.ops.mesh.select_all(action='SELECT')
    # 清除孤立无意义的 seam
    bpy.ops.uv.seams_from_islands() # 根据现有良好大岛重新设定 seam
    
    # 2. 重新进行平滑展开
    bpy.ops.uv.unwrap(method='ANGLE_BASED', margin=0.002)
    bpy.ops.uv.average_islands_scale()
    bpy.ops.uv.pack_islands(margin=0.002)
    
    bmesh.update_edit_mesh(obj.data)
    bpy.ops.object.mode_set(mode='OBJECT')

# 在 ZEN UV 展开后运行 cleanup
cleanup_fragment_uv_islands(bpy.context.active_object)

```

### 一、完整脚本：`scripts/uv_pipeline.py`

在项目中可以直接运行：
`blender --background v5_run/03_remesh.blend --python scripts/uv_pipeline.py`

```python
import os
import sys
import subprocess
import bpy

# ==================== 1. 配置路径信息 ====================
# RizomUV 可执行文件路径
RIZOM_EXE_PATH = r"D:\Program Files\Rizom Lab\RizomUV 2025.0\rizomuv.exe"

# 工作目录与中间文件路径
WORK_DIR = os.path.abspath("v5_run")
IN_FBX_PATH = os.path.join(WORK_DIR, "in_for_rizom.fbx").replace("\\", "/")
OUT_FBX_PATH = os.path.join(WORK_DIR, "out_from_rizom.fbx").replace("\\", "/")
LUA_SCRIPT_PATH = os.path.join(WORK_DIR, "auto_rizom_unwrap.lua").replace("\\", "/")
SAVE_BLEND_PATH = os.path.join(WORK_DIR, "04_uv.blend")


# ==================== 2. 动态生成 RizomUV LUA 脚本 ====================
def generate_rizom_lua_script(lua_path, input_fbx, output_fbx):
    """
    生成利用 UVBorders 传递 Seam 的 RizomUV LUA 自动化脚本
    """
    lua_content = f"""-- RizomUV Border-Seam Pipeline Script
-- 1. 加载 FBX 模型并重置基础 UV 坐标
ZomLoad({{File={{Path="{input_fbx}", ImportGroups=true, XYZ=true}}}})

-- 2. 核心突破：直接抓取 Blender 导出的基础 UV 岛边界（即 Mark Seam 处）
ZomSelect({{PrimType="Edge", WorkingSet="UVBorders", ResetBefore=true}})

-- 3. 将选中边缘转换为 3D 物理缝合线 (Cut)
ZomCut({{PrimType="Edge", WorkingSet="Selected"}})

-- 4. 执行工业级 Unfold 与 Stretch 消除 (彻底消除拉伸)
ZomUnfold({{PrimType="Polygon", WorkingSet="All", Iterations=20, Roomspace=0.005, PinIslandCenters=false}})
ZomOptimize({{Iterations=15, Method="Mosaic"}})

-- 5. 高利用率 Packing 与像素密度对齐
ZomIslandGroup({{Mode="Set", WorkingSet="All", PackProperties={{AlignMode=1}}}})
ZomPack({{ProcessTile=0, MarginVisible=true, MarginPS=0.002, Rotate={{Mode=2, Step=90}}}})

-- 6. 保存导出并退出
ZomSave({{File={{Path="{output_fbx}"}}}})
ZomQuit()
"""
    with open(lua_path, "w", encoding="utf-8") as f:
        f.write(lua_content)
    print(f"[UV Pipeline] LUA 脚本已成功生成: {lua_path}")


# ==================== 3. 主运行流程 ====================
def run_uv_pipeline():
    print("[UV Pipeline] 启动全自动展 UV 管线...")

    # A. 获取活动或低模对象 (90K QuadRemesher 低模)
    obj = bpy.context.active_object
    if not obj or obj.type != 'MESH':
        # 若无选中对象，默认寻找名称中带 'remesh' 的网格
        for o in bpy.data.objects:
            if o.type == 'MESH' and ("remesh" in o.name.lower() or "low" in o.name.lower()):
                obj = o
                break
    
    if not obj:
        raise RuntimeError("未能在场景中找到有效的低模网格对象！")

    print(f"[UV Pipeline] 处理目标模型: {obj.name} (面数: {len(obj.data.polygons)})")

    # B. 进入编辑模式，执行基础 unwrap 将 Mark Seam 转换为 UV Borders
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.select_all(action='SELECT')
    
    # 执行极简基础展开，将 Edge Seam 转化为实体 UV 岛边界
    bpy.ops.uv.unwrap(method='ANGLE_BASED', margin=0.001)
    bpy.ops.object.mode_set(mode='OBJECT')

    # C. 导出包含基础 UV Borders 的 FBX
    os.makedirs(WORK_DIR, exist_ok=True)
    bpy.ops.wm.obj_export(filepath=IN_FBX_PATH) if sys.version_info >= (3, 10) and hasattr(bpy.ops.wm, "obj_export") else None
    
    # 强制选中当前对象导出 FBX
    bpy.ops.object.select_all(action='DESELECT')
    obj.select_set(True)
    
    bpy.ops.export_scene.fbx(
        filepath=IN_FBX_PATH,
        use_selection=True,
        mesh_smooth_type='FACE',
        add_leaf_bones=False,
        use_triangles=False
    )
    print(f"[UV Pipeline] 已导出中间 FBX 到: {IN_FBX_PATH}")

    # D. 动态生成并调用 RizomUV 执行后台处理
    generate_rizom_lua_script(LUA_SCRIPT_PATH, IN_FBX_PATH, OUT_FBX_PATH)

    if not os.path.exists(RIZOM_EXE_PATH):
        raise FileNotFoundError(f"未找到 RizomUV 主程序，请检查路径: {RIZOM_EXE_PATH}")

    rizom_cmd = [
        RIZOM_EXE_PATH,
        "/cfi", LUA_SCRIPT_PATH, # 执行指定 LUA 脚本
        "/nu",                   # 无 GUI 静默运行
        "/nle"                   # 处理完毕后自动退出
    ]
    
    print("[UV Pipeline] 正在调用 RizomUV 后台解算中...")
    result = subprocess.run(rizom_cmd, capture_output=True, text=True)
    if result.returncode != 0:
        print(f"[RizomUV Error] {result.stderr}")
        raise RuntimeError("RizomUV 处理失败，请检查输出日志！")
    
    print("[UV Pipeline] RizomUV 解算完成！")

    # E. 重新导入携带完美 UV 的 FBX 并覆盖当前对象的 UV 数据
    if not os.path.exists(OUT_FBX_PATH):
        raise FileNotFoundError(f"未找到 RizomUV 输出结果: {OUT_FBX_PATH}")

    # 导入处理完的 FBX
    bpy.ops.import_scene.fbx(filepath=OUT_FBX_PATH)
    imported_obj = [o for o in bpy.context.selected_objects if o != obj][0]

    # 将新 FBX 的 UV Map 复制/传递给原始低模对象
    uv_src_layer = imported_obj.data.uv_layers.active.data
    
    # 确保原对象有 UV 层
    if not obj.data.uv_layers:
        obj.data.uv_layers.new(name="UVMap")
    uv_dst_layer = obj.data.uv_layers.active.data

    # 逐 Loop 传递坐标
    for i, loop in enumerate(imported_obj.data.loops):
        uv_dst_layer[i].uv = uv_src_layer[i].uv

    # 清理导入的临时 FBX 模型
    bpy.data.objects.remove(imported_obj, do_unlink=True)

    print("[UV Pipeline] UV 坐标已完美成功回传至原始模型！")

    # F. 保存处理后的工程文件 04_uv.blend
    bpy.ops.wm.save_as_mainfile(filepath=SAVE_BLEND_PATH)
    print(f"[UV Pipeline] 工程已保存至: {SAVE_BLEND_PATH}")


if __name__ == "__main__":
    run_uv_pipeline()

```

### 二、原理与执行细节说明

1. **避免 Edge ID 错位**：脚本先在 Blender 中进行了一次秒级的基础 `unwrap()`。此步骤将你标记的 `Seam` 直接转化为模型的 `UV Borders`（UV 边界）。


2. **传递与切割**：FBX 格式会完整保留 UV 坐标。RizomUV 启动后直接运行 `ZomSelect({PrimType="Edge", WorkingSet="UVBorders"})` 抓取 UV 边界并转换为 3D 切割线（Cut），规避了传递 Edge ID 数组导致的错位。


3. **消除拉伸与碎片**：在 LUA 中调用了 `ZomUnfold({Iterations=20})` 和 `ZomOptimize({Method="Mosaic"})`，强制进行 20 次松弛迭代，大幅提升 UV 展开质量并消解 2000 多个碎块岛。
